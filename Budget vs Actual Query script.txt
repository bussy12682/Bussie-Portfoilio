-- Q1 - Are we spending within budget, and how is our budget performance changing over time
SELECT
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND(
        (SUM(Budget_Amount) - SUM(Actual_Amount))
        / NULLIF(SUM(Budget_Amount), 0) * 100,
        2
    ) AS variance_pct
FROM budgetvsactual_clean;

-- Performance over time (monthly)
SELECT YEAR(Date) AS year, MONTH(Date) AS month,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
GROUP BY YEAR(Date), MONTH(Date)
ORDER BY year, month;

-- Performance over time (yearly)
SELECT YEAR(Date) AS year,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
GROUP BY YEAR(Date)
ORDER BY year;

-- Okay, but when exactly did we spend significantly more than planned?
SELECT YEAR(Date) AS year, MONTH(Date) AS month,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance
FROM budgetvsactual_clean
GROUP BY YEAR(Date), MONTH(Date)
ORDER BY variance ASC
LIMIT 5;

-- Q2 - Which department are over spending and underspending?
SELECT Department,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
GROUP BY Department
ORDER BY variance ASC;

-- Did the same departments drive the overspending every year, or did the problem shift between departments?
SELECT YEAR(Date) AS year, Department,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
WHERE Department <> 'Unknown'
GROUP BY YEAR(Date), Department
ORDER BY year, variance ASC;

-- Q3 - Which expense categories are contributing most to overspending?
SELECT Category,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
WHERE Category <> 'Unknown'
GROUP BY Category
ORDER BY variance ASC;

-- Which departments are responsible for these high-cost categories?
SELECT Department, Category,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance
FROM budgetvsactual_clean
WHERE Department <> 'Unknown' AND Category <> 'Unknown'
GROUP BY Department, Category
ORDER BY variance ASC;

-- Q4 - Which regions have the best and worst budget performance?
SELECT Region,
    SUM(Budget_Amount) AS total_budget,
    SUM(Actual_Amount) AS total_actual,
    SUM(Budget_Amount) - SUM(Actual_Amount) AS variance,
    ROUND((SUM(Budget_Amount) - SUM(Actual_Amount)) / NULLIF(SUM(Budget_Amount), 0) * 100, 2) AS variance_pct
FROM budgetvsactual_clean
WHERE Region <> 'Unknown'
GROUP BY Region
ORDER BY variance ASC;
